ROOT = pathlib.Path(__file__).resolve().parents[5]

assert limit in (600, 2700, 20400)

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def binding():
    names = subprocess.run(['git', 'ls-files', '-z'], cwd=ROOT, capture_output=True, check=True).stdout.decode().split('\0')
    paths = {ROOT / n for n in names if n}
    for folder in ['rust', 'docs/rust-tauri/R06/repair-current', 'artifacts/rust-tauri/R06/AR/ENTRY-01', 'artifacts/rust-tauri/R06/AR/SCOPE-01', 'artifacts/rust-tauri/R06/AR/W03-PREP-01', 'Lingxi_Rust_Tauri_Taskbooks_2026-09-23']:
        paths.update(p for p in (ROOT / folder).rglob('*') if p.is_file() and not any(q.startswith('target') or q == 'node_modules' for q in p.relative_to(ROOT).parts))
    paths.add(pathlib.Path(__file__).resolve())
    paths.add(ROOT / 'AGENTS.md')
    paths.update(BASE.glob('*.py'))
    return {str(p.relative_to(ROOT)): sha(p) for p in sorted(paths) if p.is_file()}

def external_binding():
    if '--manifest-path' not in argv:
        return {}
    root = (ROOT / argv[argv.index('--manifest-path') + 1]).resolve().parent
    if root == ROOT / 'rust':
        return {}
    return {str(p): sha(p) for p in sorted(root.rglob('*')) if p.is_file() and not any(q.startswith('target') or q == 'node_modules' for q in p.relative_to(root).parts)}

def bins():
    paths = list((target / 'debug/deps').glob('*')) + list((target / 'debug').glob('*'))
    return {str(p): sha(p) for p in paths if p.is_file() and os.access(p, os.X_OK) and not p.name.endswith('.dylib')}

def processes():
    result = subprocess.run(['ps', '-axo', 'pid=,ppid=,pgid=,rss=,command='], capture_output=True, text=True, check=True)
    table = {}
    for line in result.stdout.splitlines():
        a = line.split(None, 4)
        if len(a) == 5:
            table[int(a[0])] = (*map(int, a[1:4]), a[4])
    return table

env.update(PATH='/Users/study_superior/.cargo/bin:' + env.get('PATH', ''), LINGXI_HOME=str(home), TMPDIR=str(tmp) + '/', LINGXI_W03_EVIDENCE=str(out), TEST_MEM_LIMIT_MB='8192', TEST_TIME_LIMIT_SEC='600', LINGXI_TEST_GUARD='1', TEST_GUARD_VERBOSE='1', CARGO_BUILD_JOBS='8', CARGO_TARGET_DIR=str(target))

(out / 'run-start.json').write_text(json.dumps(r, indent=2))
seen = {}
actual_bins = {}
maxrss = 0
p = None
interrupted = None

def stop(signum, frame):
    global interrupted
    interrupted = signum

signal.signal(signal.SIGTERM, stop)
signal.signal(signal.SIGINT, stop)

def terminate_owned(table, owned):
    for pid in sorted(owned, reverse=True):
        try:
            os.kill(pid, signal.SIGTERM)
        except ProcessLookupError:
            pass
    try:
        p.wait(timeout=5)
    except subprocess.TimeoutExpired:
        os.killpg(p.pid, signal.SIGKILL)
        p.wait()
    time.sleep(.1)
    survivors = {pid for pid, v in processes().items() if v[1] == p.pid}
    for pid in survivors:
        try:
            os.kill(pid, signal.SIGKILL)
        except ProcessLookupError:
            pass

with (out / 'stdout.log').open('w') as stdout, (out / 'stderr.log').open('w') as stderr:
    p = subprocess.Popen(argv, cwd=ROOT, env=env, stdout=stdout, stderr=stderr, start_new_session=True)
    r['child_pid'] = p.pid
    while p.poll() is None:
        table = processes()
        owned = {p.pid}
        while True:
            new = owned | {pid for pid, v in table.items() if v[0] in owned or v[1] == p.pid}
            if new == owned:
                break
            owned = new
        rss = sum(table[pid][2] for pid in owned if pid in table)
        maxrss = max(maxrss, rss)
        for pid in owned:
            if pid not in table:
                continue
            v = table[pid]
            if pid not in seen:
                seen[pid] = {'pid': pid, 'ppid': v[0], 'pgid': v[1], 'command': v[3], 'first_seen': time.time()}
                (out / 'processes.json').write_text(json.dumps(list(seen.values()), indent=2))
            executable = pathlib.Path(v[3].split(' ', 1)[0])
            if executable.is_absolute() and executable.is_file() and str(target) in str(executable) and str(executable) not in actual_bins:
                actual_bins[str(executable)] = {'before_execution_observation': sha(executable), 'pid': pid, 'first_seen': time.time()}
        if interrupted or rss > 8192 * 1024 or time.time() - r['start_time'] > limit:
            r['guard_trigger'] = 'signal' if interrupted else ('rss' if rss > 8192 * 1024 else 'timeout')
            r['signal'] = interrupted
            terminate_owned(table, owned)
            break
        time.sleep(.5)
    r['exit_code'] = p.wait()

r['actual_executed_bins'] = actual_bins
